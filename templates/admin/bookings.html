The query behind this list 
-- One search box over five columns, so an agent can type whatever the
-- passenger happens to remember. ILIKE is case-insensitive; the EXISTS
-- reaches into the booking's tickets without multiplying the rows.
SELECT b.pnr, b.booked_at, b.status, p.full_name, p.email,
       count(t.ticket_id) FILTER (WHERE t.status <> 'CANCELLED') AS pax,
       sum(t.fare_paid)   FILTER (WHERE t.status <> 'CANCELLED') AS value,
       (SELECT sum(amount) FROM payment
         WHERE booking_id = b.booking_id)                        AS paid,
       string_agg(DISTINCT f.flight_no, ' ')                     AS flights
  FROM booking b
  JOIN passenger p ON p.passenger_id = b.passenger_id
  LEFT JOIN ticket t ON t.booking_id = b.booking_id
  LEFT JOIN flight f ON f.flight_id  = t.flight_id
 WHERE b.pnr ILIKE %s OR p.full_name ILIKE %s OR p.email ILIKE %s
    OR EXISTS (SELECT 1 FROM ticket t2 JOIN flight f2 USING (flight_id)
                WHERE t2.booking_id = b.booking_id
                  AND (t2.ticket_no ILIKE %s OR f2.flight_no ILIKE %s))
 GROUP BY b.booking_id, p.full_name, p.email
 ORDER BY b.booked_at DESC
 LIMIT 25 OFFSET %s;

-- value is what the live tickets are worth; paid is the payment ledger,
-- which nets refunds off. They differ exactly when something was cancelled.
