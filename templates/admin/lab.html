Defence 1 — the row lock 
-- Taken first, before anything is read. ORDER BY seat_id so two
-- multi-seat bookings cannot deadlock by locking in opposite orders.
SELECT seat_id FROM seat
 WHERE seat_id = ANY(%s)
 ORDER BY seat_id
   FOR UPDATE;

-- T1 gets the lock. T2 *blocks on this statement* rather than reading a
-- stale "seat is free". When T1 commits, T2 resumes, re-reads, finds the
-- seat gone, and bails cleanly — no error, no duplicate, no lost update.
Defence 2 — the declarative backstop 
CREATE UNIQUE INDEX uq_ticket_seat_per_flight
    ON ticket (flight_id, seat_id)
 WHERE seat_id IS NOT NULL;

-- PARTIAL, so cancelling a ticket (seat_id -> NULL) frees the seat for the
-- next passenger while two live tickets still cannot share it.
--
-- This holds even with the lock switched off, at a weaker isolation level,
-- or from a different client entirely. A correctness guarantee should never
-- depend on a developer remembering something.
